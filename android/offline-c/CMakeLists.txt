cmake_minimum_required(VERSION 3.22.1)
project(myct_offline_c LANGUAGES C CXX ASM)
if(NOT DEFINED MYCT_VENDOR)
    message(FATAL_ERROR "Run prepare.py and set MYCT_VENDOR")
endif()
if(ANDROID)
    set(WAMR_BUILD_PLATFORM "android")
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        set(WAMR_BUILD_TARGET "AARCH64")
    elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
        set(WAMR_BUILD_TARGET "ARMV7")
    elseif(ANDROID_ABI STREQUAL "x86_64")
        set(WAMR_BUILD_TARGET "X86_64")
    else()
        message(FATAL_ERROR "Unsupported ABI")
    endif()
else()
    set(WAMR_BUILD_PLATFORM "linux")
    set(WAMR_BUILD_TARGET "X86_64")
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 0)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 1)
set(WAMR_BUILD_SHARED_MEMORY 0)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_BUILD_LIB_WASI_THREADS 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_DISABLE_STACK_HW_BOUND_CHECK 1)
set(WAMR_ROOT_DIR ${MYCT_VENDOR}/wamr)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(wcpl STATIC ${MYCT_VENDOR}/wcpl/c.c ${MYCT_VENDOR}/wcpl/l.c ${MYCT_VENDOR}/wcpl/p.c ${MYCT_VENDOR}/wcpl/w.c)
set_source_files_properties(${MYCT_VENDOR}/wcpl/c.c PROPERTIES COMPILE_DEFINITIONS "main=wcpl_main")
# Upstream predates C23 const-correct generic string APIs in glibc 2.43.
set_target_properties(wcpl PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
add_executable(myct-c-worker worker.cpp ${WAMR_RUNTIME_LIB_SOURCE})
target_compile_features(myct-c-worker PRIVATE cxx_std_20)
target_compile_options(myct-c-worker PRIVATE -fPIE -ffunction-sections -fdata-sections)
target_link_options(myct-c-worker PRIVATE -pie -Wl,--gc-sections -Wl,-z,noexecstack -Wl,-z,max-page-size=16384)
target_link_libraries(myct-c-worker wcpl m dl)
if(ANDROID)
    target_link_libraries(myct-c-worker log)
else()
    target_link_libraries(myct-c-worker pthread)
endif()
# Android PackageManager extracts lib/<abi>/lib*.so into a read-only executable
# install directory. This is a PIE worker shipped in the APK, never a generated
# user executable. User artifacts are .wasm DATA and are interpreted only.
set_target_properties(myct-c-worker PROPERTIES OUTPUT_NAME "libmyct-c-worker.so")
install(TARGETS myct-c-worker RUNTIME DESTINATION .)
